SELECT `order details`.`OrderID`, (`order details`.`Quantity` * `order details`.`UnitPrice`) * (1 - `order details`.`Discount`) AS `productTotal`

FROM `order details`

WHERE `order details`.`OrderID` = 10250



SELECT `order details`.`OrderID`, round((`order details`.`Quantity` * `order details`.`UnitPrice`) * (1 - `order details`.`Discount`), 4) AS `productTotal`

FROM `order details`

WHERE `order details`.`OrderID` = 10250




SELECT `order details`.`OrderID`, round(SUM((`order details`.`Quantity` * `order details`.`UnitPrice`) * (1 - `order details`.`Discount`)),4) AS `OrderTotal`

FROM `order details`

WHERE 1
