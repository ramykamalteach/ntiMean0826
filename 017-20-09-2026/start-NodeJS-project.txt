create node project
>npm init -y

run project
>node fileName.ext
>node index.js


install nodemon locally
>npm install --save-dev nodemon

install nodemon globally
>npm install -g nodemon

run project by nodemon
>nodemon app.js

we can change package.json by adding
"scripts": {
  "start": "node app.js",
  "dev": "nodemon app.js"
}
then run project by
>npm run dev

stop nodemon in terminal by : ctrl+c

non-blocking concept :

console.log("Hello, World!");

const fs = require('fs');

fs.readFile("file.txt", (err, data) => {
  if (err) throw err;
  console.log("File has been read");
});

import file to our code.

Modules: 
    - core (dependency)
    - npm packages
    - local modules

- core (dependency) importing ex:
	const fs = require('fs');
	const path = require('path');
	const http = require('http');

- npm packages importing ex:
	install using npm >npm install express
	import it in code :
		const express = require('express');

- local modules:
	the module opSum.js :
		const sum = (a, b) => {
    			return a + b;
		}
		module.exports = {
    			"summation" : sum
		}
	
	import in a file
		const sum01 = require("./opSum");
		console.log(sum01.summation(11111, 33));


*** we can use import for : core, npm, local modules also
	import fs from 'fs';
	import express from 'express';
	import myUtils from './opSum.js'; 

----------
ex for local
	the module opSum.js :
		const sum = (a, b) => {
    			return a + b;
		}
		export const summation = sum;
	
	import in a file
		import { summation } from "./opSum.js";
		console.log(summation(3, 33));
---------------
use import in node project :
	- change
		"type": "module"
	in package.json

	- change export formula
		export default {
    			summation: sum,
    			subtract: subtract
		};

	- import in another module :

	import calculate from "./opSum.js"; 

	let result = calculate.summation(11, 22);
	console.log(result);
	result = calculate.subtract(11, 22);
	console.log(result);


other method :
	in calc.js

		export const summation = (a, b) => {
			return a + b;
		};

		export const subtract = (a, b) => {
			return a - b;
		};

	import in other module :
		import { summation, subtract } from './calc.js';

		let result = summation(11, 22);
		console.log(result);
		result = subtract(100, 5);
		console.log(result);
